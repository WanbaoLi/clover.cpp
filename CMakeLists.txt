cmake_minimum_required(VERSION 3.18)
project(CloverSVE VERSION 1.0.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)
set(CLOVER_QXS "general" CACHE STRING "QXS implementation: general or sve512")
set_property(CACHE CLOVER_QXS PROPERTY STRINGS general sve512)
option(CLOVER_OPENACC "Compile the optional Bridge++ OpenACC library backend" OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build on 64-bit Linux (including WSL2), or cross-compile for Linux.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This distribution requires a 64-bit toolchain.")
endif()

add_library(lime STATIC
  extra/lime/lib/lime_fseeko.c extra/lime/lib/lime_header.c extra/lime/lib/lime_reader.c
  extra/lime/lib/lime_utils.c extra/lime/lib/lime_writer.c)
target_include_directories(lime PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/extra/lime/include> $<INSTALL_INTERFACE:include/lime>)
target_compile_definitions(lime PUBLIC _FILE_OFFSET_BITS=64)

file(GLOB_RECURSE bridge_sources CONFIGURE_DEPENDS
  "src/lib/*.cpp" "src/lib_alt/*.cpp" "src/lib_alt_QXS/*.cpp")
# The optimized and original versions of these two kernels export identical symbols.
list(FILTER bridge_sources EXCLUDE REGEX "/(Force/Fermion|Measurements/Fermion)/Org/")
add_library(bridge STATIC ${bridge_sources})
target_compile_features(bridge PUBLIC cxx_std_11)
target_include_directories(bridge PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/lib>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/lib_alt> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/lib_alt_QXS>
  $<INSTALL_INTERFACE:include/bridge> $<INSTALL_INTERFACE:include/bridge/lib>
  $<INSTALL_INTERFACE:include/bridge/lib_alt> $<INSTALL_INTERFACE:include/bridge/lib_alt_QXS>)
target_link_libraries(bridge PUBLIC MPI::MPI_CXX OpenMP::OpenMP_CXX lime m)
target_compile_definitions(bridge PUBLIC USE_LIMELIB USE_MPI USE_IMP USE_GROUP_SU3
  USE_OPENMP USE_STD_COMPLEX LIB_CPP11 USE_FACTORY USE_ALT_CODE USE_ALT_QXS MPICH_SKIP_MPICXX OMPI_SKIP_MPICXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_definitions(bridge PUBLIC PC_CLANG)
else()
  target_compile_definitions(bridge PUBLIC PC_GNU)
endif()
if(CLOVER_QXS STREQUAL "sve512")
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_FLAGS "-march=armv8.2-a+sve+sve2 -msve-vector-bits=512")
  check_cxx_source_compiles("#include <arm_sve.h>\nstatic_assert(__ARM_FEATURE_SVE_BITS==512, \"SVE512\");\nint main(){return svcntd()!=8;}" CLOVER_CAN_BUILD_SVE512)
  unset(CMAKE_REQUIRED_FLAGS)
  if(NOT CLOVER_CAN_BUILD_SVE512)
    message(FATAL_ERROR "sve512 requires an AArch64 C++ compiler with 512-bit ACLE SVE2 support.")
  endif()
  target_compile_definitions(bridge PUBLIC CLOVER_QXS_SVE512)
  target_compile_options(bridge PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-march=armv8.2-a+sve+sve2;-msve-vector-bits=512>)
elseif(NOT CLOVER_QXS STREQUAL "general")
  message(FATAL_ERROR "Use -DCLOVER_QXS=general or sve512.")
endif()
if(CLOVER_OPENACC)
  find_package(OpenACC REQUIRED COMPONENTS CXX)
  file(GLOB_RECURSE accel_sources CONFIGURE_DEPENDS "src/lib_alt_Accel/*.cpp")
  target_sources(bridge PRIVATE ${accel_sources})
  target_compile_definitions(bridge PUBLIC USE_ALT_ACCEL USE_ACCEL_OPENACC)
  target_include_directories(bridge PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/lib_alt_Accel> $<INSTALL_INTERFACE:include/bridge/lib_alt_Accel>)
  target_link_libraries(bridge PUBLIC OpenACC::OpenACC_CXX)
endif()

if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(clover-base-check tests/base_operator.cpp)
  target_link_libraries(clover-base-check PRIVATE bridge)
  add_test(NAME analytic_clover_plane_wave COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 1 ${MPIEXEC_PREFLAGS} $<TARGET_FILE:clover-base-check> ${MPIEXEC_POSTFLAGS})
  set_tests_properties(analytic_clover_plane_wave PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1" TIMEOUT 60)
endif()
install(TARGETS bridge lime EXPORT CloverSVETargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/ DESTINATION include/bridge FILES_MATCHING PATTERN "*.h")
install(DIRECTORY extra/lime/include/ DESTINATION include/lime FILES_MATCHING PATTERN "*.h")
install(FILES LICENSE.txt DESTINATION share/clover-sve/licenses)
install(FILES extra/lime/COPYING DESTINATION share/clover-sve/licenses RENAME C-LIME-COPYING)
install(FILES README.md SOURCES.md DESTINATION share/clover-sve)
configure_package_config_file(cmake/CloverSVEConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/CloverSVEConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/CloverSVE")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CloverSVEConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(EXPORT CloverSVETargets NAMESPACE CloverSVE:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/CloverSVE")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CloverSVEConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/CloverSVEConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/CloverSVE")
